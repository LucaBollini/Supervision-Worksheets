\begin{center}
\huge{\textbf{Subsequences}}
\end{center}

\begin{enumerate}
    \item Let $x_n$ and $y_n$ both be bounded sequences. Prove that there is a sequence of integers $n_k\to\infty$ such that $x_{n_k}$ and $y_{n_k}$ both converge.
    \item Prove that $a_n$ is bounded if and only if every subsequence of $a_n$ has a convergent subsequence.
    \item Let $x_n$ be a sequence of real numbers that lie in the interval $[0,1]$. Suppose that
    \[
    x_{n+1}-x_n\to 0 \quad \text{as}\quad n\to \infty,
    \]
    and any convergent subsequence of $x_n$ either converges to $0$ or $1$. Prove that the entire sequence converges.
\end{enumerate}

\begin{center}
\huge{\textbf{Group Isomorphisms}}
\end{center}

\begin{enumerate}
    \item Consider the groups $(\{2,4,6,8\}, \times)$ with multiplication modulo $10$, and $(\{1,3,5,7\}, \times)$ with multiplication modulo $8$. Are these two groups isomorphic?
    \item For any group $G$, an isomorphism $\varphi:G\to G$ is called an \emph{automorphism}. Prove that the set of automorphisms on $G$ is itself a group with respect to function composition.
\end{enumerate}
