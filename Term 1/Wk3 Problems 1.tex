\huge{\textbf{Group Theory}}
\end{center}

\begin{enumerate}
    \item For each of the following groups $(G, \star)$ and subsets $H\subset G$, determine whether or not $(H,\star)$ is a   subgroup. Justify your answers.
    \begin{enumerate}[(i)]
    \item $G= 3\Z$, $\star=$ addition, $H= 9\Z$
    \item $G = \R\setminus\{0\}$, $\star=$ multiplication, $H = \{2^n:n\in\Z\}$
    \item $G = \R^2$, $\star$ = vector addition, $H = \{(x,y)^T : x-y\in\Z\}$
    \end{enumerate}
        
    \item For each of the following groups, $G$, determine the order of the element $g$ and write down the cyclic subgroup generated by it, $\langle g\rangle$.
    \begin{enumerate}[(i)]
        \item $G=\Z/7\Z, g=[2]_7$
        \item $G=\Z/12\Z, g=[8]_{12}$
        \item $G= \text{GL}_2(\R)$, $g = \begin{pmatrix}-\frac{1}{2} &-\frac{\sqrt3}{2}\\ \frac{\sqrt3}{2} &-\frac{1}{2}\end{pmatrix}$
        \item $G= \text{GL}_2(\R)$, $g = \begin{pmatrix}0 &-2\\ 2 & 0\end{pmatrix}$
    \end{enumerate}
    
    \item Prove that any group with a prime number of elements is cyclic.
    
\end{enumerate}
