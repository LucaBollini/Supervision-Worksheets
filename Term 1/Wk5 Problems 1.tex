\begin{center}
\huge{\textbf{Permutations}}
\end{center}

\begin{enumerate}
    \item Find the inverse of the following permutations and write your answer as a product of disjoint cycles.
    \begin{enumerate}[(i)]
        \item (1~2~4)(3~5~7~6)
        \item (1~2)(2~4~3)(2~3~5)
    \end{enumerate}
    \item Find all possible orders of elements in $S_7$. How many elements are there with order $2$?
    \item Let $\sigma$ be a $k$-cycle. If $m$ is a positive integer, prove that $\sigma^m$ is a $k$-cycle if and only if $k$ and $m$ are coprime.
\end{enumerate}

\begin{center}
\huge{\textbf{Applying Lagrange's Theorem}}
\end{center}

\begin{enumerate}
    \item Show that every group of order $55$ contains an element of order $5$ and an element of order $11$.
    \item Let $G$ be a (possibly infinite) group with subgroups $H$ and $K$ such that $K$ is also a subgroup of $H$. Suppose that $[G : H]=m$ and $[H : K]=n$. Show that $[G : K]=mn$.
\end{enumerate}
