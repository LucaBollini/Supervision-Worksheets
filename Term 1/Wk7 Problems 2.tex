        \underline{Solution.}

        \[
        \begin{aligned}
            &hH=H\quad\forall h\in H\\
            &gH=H'\quad\forall g\notin H \because 1\in H
        \end{aligned}
        \]
        Since $1\notin H'\ g^{-1}\notin H.\ \therefore\ \exists h\in H$ st
        \[
        \begin{aligned}
            &gh= g^{-1}\\
            &g^2=h^{-1}\in H\quad \forall g\in G.
        \end{aligned}
        \]
        So $(a~b~c)^2\in H \implies \big((a~b~c)^2\big)^2=(a~b~c)\in H$. So $H$ contains at least $8$ elements $\lightning$.

\begin{center}
\huge{\textbf{Ordering}}
\end{center}

\begin{enumerate}
    \item Define the relation $\preceq$ on the set of all real valued polynomials $\R[x]$ by
    \[
    P\preceq Q \iff P(x)\leq Q(x) \text{ for all } x\in[0,1].
    \]
    Prove that $(\R[x],\preceq)$ is a partially ordered set (poset).
    \item Let
    \[
    A:= \{(m,n):m,n\in\N\text{ and }\gcd(m,n)=1\},
    \]
    and define the relation $\preceq$ on $A$ with
    \[
    (a,b)\preceq (c,d) \iff ad\leq bc.
    \]
    Prove that $(A,\preceq)$ is a totally ordered set.
\end{enumerate}
