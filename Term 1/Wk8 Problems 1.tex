\begin{center}
\huge{\textbf{Ideals}}
\end{center}

\begin{enumerate}
    \item Let $I_1\subseteq I_2\subseteq I_3\subseteq\cdots$ be a sequence of ideals in a ring $R$. Define
    \[
    U := \{a: a\in R\text{ and } a\in I_n \text{ for some } n\}.
    \]
    Prove that $U$ is an ideal of $R$.
    \item Let $R$ be a commutative ring with unity, and let $P\subset R$ be a \emph{prime ideal}, by which we mean an ideal such that if $ab\in P$ then either $a\in P$, $b\in P$ or both.
    
    If there exist ideals $I,J\subseteq R$ with $P\subseteq I \cup J$, is it necessarily true that either $P\subseteq I$, $P\subseteq J$ or both?
\end{enumerate}

\begin{center}
\huge{\textbf{Convergence Tests}}
\end{center}

\begin{enumerate}
    \item Show that if $\series a_n$ and $\series b_n$ are convergent series of non-negative terms, then so is $\series\max(a_n,b_n)$.
    \item Investigate the convergence of
    \[
    \series \left(\frac1{\sqrt{n}}-\frac1{\sqrt{n+1}}\right).
    \]
    \item Investigate the convergence of
    \[
    \series \left(\frac{3n+1}{4-2n}\right)^{2n}.
    \]
    \item Show that for each fixed $k\in\R$ and $a\in(0,1)$, the series $\series n^k a^n$ converges.
\end{enumerate}
