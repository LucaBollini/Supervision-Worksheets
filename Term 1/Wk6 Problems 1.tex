\begin{center}
\huge{\textbf{ODEs by Substitution}}
\end{center}

\begin{enumerate}
    \item Using the substitution $z(x) = 1/y(x)$, find the general solution to
    \[
    xy' + y = y^2\log x.
    \]
    \item Using the substitution $z(x) = 1/y(x)^2$, solve
    \[
    y' + xy = y^3e^{x^2}
    \]
    subject to the initial condition $y=\frac{1}{2}$ when $x=0$.
\end{enumerate}

\begin{center}
\huge{\textbf{Euclidean Algorithm}}
\end{center}

\begin{enumerate}
    \item Find the greatest common divisor (gcd) of $2244$ and $2145$
    \item Find integers $x$ and $y$ such that
    \[
    119x + 34y = \gcd(119,34)
    \]
    \item An exhibit in an art gallery consists of square ceramic tiles in a $901$ by $629$ grid. A laser is directed across the exhibit, from the top left corner of the grid to the bottom right. Exactly how many tiles does the laser pass over?
    % The answer is 901 + 629 - gcd = 1513
\end{enumerate}
